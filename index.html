function renderTransaksi() { const container = document.getElementById('listTransaksi'); if (!container) return; const f = (document.getElementById('filterGeneral')?.value || "").toLowerCase(); const list = Array.isArray(dataTransaksi) ? dataTransaksi : []; const filtered = list.filter(t => (t && t.pembeli || "").toLowerCase().includes(f) || (t && t.barang || "").toLowerCase().includes(f) ); let html = ""; filtered.slice().reverse().forEach(t => { if (!t) return; // PENJAGA: Biar toLocaleString gak bikin macet login const h = (typeof t.harga === 'number') ? t.harga : (parseFloat(t.harga) || 0); const s = (typeof t.sisa === 'number') ? t.sisa : (parseFloat(t.sisa) || 0); html += ` ${t.tglDisplay || '-'}
${t.pembeli || 'Tanpa Nama'} ${t.alamat || '-'}
Rp ${h.toLocaleString('id-ID')} Rp ${s.toLocaleString('id-ID')}
`; }); container.innerHTML = html; }
